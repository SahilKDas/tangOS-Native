cmake_minimum_required(VERSION 3.20)
project(TangOSLite VERSION 0.12.0 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "TangOS Lite currently targets Windows")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(lite_core src/core.cpp src/platform.cpp src/repository.cpp src/descriptor.cpp src/fleet.cpp src/mcp.cpp src/atlas_layout.cpp src/network.cpp src/backend.cpp src/viewer.cpp src/help.cpp src/batches.cpp)
target_include_directories(lite_core PUBLIC src)
target_link_libraries(lite_core PUBLIC bcrypt crypt32 ws2_32 winhttp)
target_compile_definitions(lite_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
if(NOT MINGW)
  message(FATAL_ERROR "The TinySkia release currently requires the MinGW GNU toolchain")
endif()
find_program(CARGO_EXECUTABLE cargo REQUIRED)
set(RASTER_TARGET "${CMAKE_CURRENT_BINARY_DIR}/raster")
set(RASTER_LIBRARY "${RASTER_TARGET}/x86_64-pc-windows-gnu/release/libtangos_raster.a")
add_custom_command(OUTPUT "${RASTER_LIBRARY}"
 COMMAND ${CMAKE_COMMAND} -E env "RUSTFLAGS=--remap-path-prefix=${CMAKE_CURRENT_SOURCE_DIR}=lite --remap-path-prefix=${CMAKE_CURRENT_BINARY_DIR}=build" ${CARGO_EXECUTABLE} build --locked --offline --release --target x86_64-pc-windows-gnu --target-dir "${RASTER_TARGET}" --manifest-path "${CMAKE_CURRENT_SOURCE_DIR}/renderer/Cargo.toml"
 DEPENDS renderer/src/lib.rs renderer/Cargo.toml renderer/Cargo.lock
 VERBATIM)
add_custom_target(raster_build DEPENDS "${RASTER_LIBRARY}")
add_library(tangos_raster STATIC IMPORTED)
set_target_properties(tangos_raster PROPERTIES IMPORTED_LOCATION "${RASTER_LIBRARY}")
add_dependencies(tangos_raster raster_build)
add_executable(TangOSLite WIN32 src/main.cpp src/skin.cpp src/console_ui.cpp src/app.rc)
target_link_libraries(TangOSLite PRIVATE lite_core comctl32 ole32 shell32 uuid dwmapi tangos_raster ws2_32 userenv ntdll)
if(MINGW)
  target_link_options(TangOSLite PRIVATE -static -municode -Wl,--no-insert-timestamp $<$<CONFIG:Release>:-s>)
  target_compile_options(lite_core PRIVATE "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=.")
  target_compile_options(TangOSLite PRIVATE "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=.")
  target_compile_options(lite_core PRIVATE -Wall -Wextra)
elseif(MSVC)
  set_property(TARGET lite_core TangOSLite PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(lite_core PRIVATE /W4 /utf-8)
  target_compile_options(TangOSLite PRIVATE /utf-8)
  target_link_options(TangOSLite PRIVATE /Brepro)
endif()
enable_testing()
add_executable(lite_tests tests/tests.cpp)
target_link_libraries(lite_tests PRIVATE lite_core)
if(MINGW)
  target_link_options(lite_tests PRIVATE -static -Wl,--no-insert-timestamp)
elseif(MSVC)
  set_property(TARGET lite_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_test(NAME core_and_disposable_git COMMAND lite_tests)
add_executable(lite_fleet_tests tests/fleet_tests.cpp src/app.rc)
target_link_libraries(lite_fleet_tests PRIVATE lite_core)
target_link_options(lite_fleet_tests PRIVATE -static -Wl,--no-insert-timestamp)
add_test(NAME fleet_descriptor_vault COMMAND lite_fleet_tests)

add_test(NAME raster_pixels COMMAND ${CARGO_EXECUTABLE} test --locked --offline --manifest-path "${CMAKE_CURRENT_SOURCE_DIR}/renderer/Cargo.toml" --target x86_64-pc-windows-gnu --target-dir "${RASTER_TARGET}/tests")

add_executable(lite_backend_tests tests/backend_tests.cpp)
target_link_libraries(lite_backend_tests PRIVATE lite_core)
target_link_options(lite_backend_tests PRIVATE -static -Wl,--no-insert-timestamp)
add_test(NAME backend_parity_services COMMAND lite_backend_tests)
