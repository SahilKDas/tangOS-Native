cmake_minimum_required(VERSION 3.20)
project(TangOSLite VERSION 0.1.0 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "TangOS Lite currently targets Windows")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(lite_core src/core.cpp src/platform.cpp src/repository.cpp)
target_include_directories(lite_core PUBLIC src)
target_link_libraries(lite_core PUBLIC bcrypt)
target_compile_definitions(lite_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_executable(TangOSLite WIN32 src/main.cpp src/app.rc)
target_link_libraries(TangOSLite PRIVATE lite_core comctl32 ole32 shell32 uuid)
if(MINGW)
  target_link_options(TangOSLite PRIVATE -static -municode -Wl,--no-insert-timestamp $<$<CONFIG:Release>:-s>)
  target_compile_options(lite_core PRIVATE "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=.")
  target_compile_options(TangOSLite PRIVATE "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=.")
  target_compile_options(lite_core PRIVATE -Wall -Wextra)
elseif(MSVC)
  set_property(TARGET lite_core TangOSLite PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(lite_core PRIVATE /W4 /utf-8)
  target_compile_options(TangOSLite PRIVATE /utf-8)
  target_link_options(TangOSLite PRIVATE /Brepro)
endif()
enable_testing()
add_executable(lite_tests tests/tests.cpp)
target_link_libraries(lite_tests PRIVATE lite_core)
if(MINGW)
  target_link_options(lite_tests PRIVATE -static -Wl,--no-insert-timestamp)
elseif(MSVC)
  set_property(TARGET lite_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_test(NAME core_and_disposable_git COMMAND lite_tests)
